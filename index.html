function renderDaftarPaketCBT(paketList) { const container = document.getElementById('paketLatihanContainer'); appState.cbtPaketList = paketList || []; if (!appState.cbtPaketList.length) { container.innerHTML = `
Belum ada paket latihan CBT yang aktif pada periode ini.
`; return; } container.innerHTML = appState.cbtPaketList.map((p, idx) => `
${escapeHtml(p.mapel)} ${p.durasi_menit} Menit

${escapeHtml(p.judul)}

Jumlah Butir Soal: ${p.jumlah_soal} Soal

`).join(''); } async function prefetchPaketLatihan() { const periodeAktif = appState.periode || 'default'; const cacheKey = `tka_cbt_paket_${periodeAktif}`; try { const url = `${API_URL}?action=latihan_soal&periode=${encodeURIComponent(appState.periode)}`; const res = await fetch(url, { method: 'GET', redirect: 'follow' }); if (!res.ok) return; const data = await res.json(); if (data && data.status === 'ok') { appState.cbtLoadedPeriode = appState.periode; appState.cbtPaketList = data.paket || []; try { localStorage.setItem(cacheKey, JSON.stringify(data.paket || [])); } catch (e) {} } } catch (e) {} } async function loadPaketLatihan() { const loader = document.getElementById('cbtLoader'); const lobby = document.getElementById('cbtLobbyWrapper'); const container = document.getElementById('paketLatihanContainer'); const periodeAktif = appState.periode || 'default'; const cacheKey = `tka_cbt_paket_${periodeAktif}`; let hasLocalCache = false; // 1. Cek apakah sudah ada di memori sesi atau localStorage if (appState.cbtLoadedPeriode === appState.periode && appState.cbtPaketList.length > 0) { loader.style.display = 'none'; lobby.style.display = 'block'; renderDaftarPaketCBT(appState.cbtPaketList); return; } try { const rawCache = localStorage.getItem(cacheKey); if (rawCache) { const parsedPaket = JSON.parse(rawCache); if (Array.isArray(parsedPaket)) { loader.style.display = 'none'; lobby.style.display = 'block'; renderDaftarPaketCBT(parsedPaket); hasLocalCache = true; } } } catch (e) {} // Tampilkan spinner hanya jika memori lokal benar-benar kosong if (!hasLocalCache) { loader.style.display = 'block'; lobby.style.display = 'none'; } // 2. Sinkronisasi paket soal terbaru di latar belakang try { const url = `${API_URL}?action=latihan_soal&periode=${encodeURIComponent(appState.periode)}`; const res = await fetch(url, { method: 'GET', redirect: 'follow' }); const data = await res.json(); appState.cbtLoadedPeriode = appState.periode; const freshPaket = data.paket || []; loader.style.display = 'none'; lobby.style.display = 'block'; // Perbarui tampilan hanya jika siswa tidak sedang mengerjakan ujian const isExamOpen = document.getElementById('cbtExamWrapper').style.display === 'block'; if (!isExamOpen) { renderDaftarPaketCBT(freshPaket); } try { localStorage.setItem(cacheKey, JSON.stringify(freshPaket)); } catch (e) {} } catch (err) { loader.style.display = 'none'; lobby.style.display = 'block'; if (!hasLocalCache) { container.innerHTML = `
Gagal memuat paket latihan soal. Silakan muat ulang halaman.
`; } } }
